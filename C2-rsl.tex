\paragraph{Solution to \cref{box:rslprod1}}
Let $f(\xi)$ be a sufficiently smooth test function. We find
\begin{align}
    &(C\otimes f)(x)\nonumber\\
    &= \int\limits_x^1 \frac{\dd z}{z} f(x/z) r(z) \cdot \left[ p(z) \right]_+\\
    &= \int\limits_0^1 \frac{\dd z}{z} f(x/z) r(z) \cdot \left[ p(z) \right]_+ - \int\limits_0^x \frac{\dd z}{z} f(x/z) r(z) \cdot \left[ p(z) \right]_+\\
    &= \int\limits_0^1\! \dd z \left(\frac{f(x/z)r(z)}{z} - f(x)r(1)\right) \cdot p(z) - \int\limits_0^x\!\dd z\, \frac{ f(x/z) r(z)}{z} \cdot p(z)\\
    &= \int\limits_x^1\! \dd z \left(\frac{f(x/z)r(z)}{z} - f(x)r(1)\right) \cdot p(z) - f(x) r(1) \int\limits_0^x\dd z~ p(z)\\
    &= \int\limits_x^1\! \dd z \left(\frac{f(x/z)(r(z)+r(1)-r(1))}{z} - f(x)r(1)\right) \cdot p(z) - f(x) r(1) \int\limits_0^x\dd z~ p(z)\\
    &= \int\limits_x^1\! \dd z \left(\frac{f(x/z)}{z} - f(x)\right) r(1)\cdot p(z) + \int\limits_x^1\! \dd z \frac{f(x/z)(r(z)-r(1)))}{z} p(z) \nonumber\\
    &\hspace{20pt}  - f(x) r(1) \int\limits_0^x\!\dd z~ p(z)\\
    &= \int\limits_x^1 \frac{\dd z}{ z} f(x/z)  r(1)\cdot \left[p(z)\right]_+ + \int\limits_x^1\! \dd z \frac{f(x/z)(r(z)-r(1)))}{z} p(z) - f(x) r(1) \int\limits_0^x\!\dd z~ p(z)
\end{align}
\begin{equation}
    \Rightarrow C^R(z) = (r(z)-r(1))p(z)\,,~  C^S(z) = r(1)p(z)\,,~ C^L(x) = -r(1)\int\limits_0^x\!\dd z\, p(z)
\end{equation}
